Experiment 08: Ethernet webserver

اهداف آزمایش

  • اتصال به اینترنت از طریق رابط اترنت با به‌کارگیری ماژول
  • میزبانی یک کارساز وب که یک صفحه وب ساده را پس از درخواست، به کارگیران تحویل دهد

پیش‌نیازها

کد کامل این آزمایش و آزمایش‌های دیگر نیز همگی در پیوست 4 آمده‌اند. در ادامه، کد این آزمایش به صورت تکه تکه توضیح داده خواهد شد.

توضیح کد آزمایش 

در کد زیر (سربرگ‌ها و تنظیمات اولیه شبکه):
•	#include  کتابخانه مخصوص ماژول ENC28J60 را بارگذاری می‌کند (کتابخانه‌ای متفاوت از کتابخانه رسمی W5100/W5500 است).
•	آرایه mac آدرس سخت‌افزاری (MAC) کارت شبکه را نگه می‌دارد. مقدار قراردادیِ نمونه در کد قابل‌تعویض با مقدار واقعی درج‌شده روی ماژول یا یک MAC تصادفی معتبر است.
•	شیء ip از نوع IPAddress آدرس آی‌پی است که سرور قرار است به آن اختصاص یابد؛ مقدار باید در رنج شبکه محلی قرار گیرد تا با سایر دستگاه‌ها تداخل نداشته باشد.
•	شیء server از کلاس EthernetServer روی پورت ۸۰ (HTTP) ساخته شده؛ این شیء به‌عنوان Listener برای کلاینت‌های TCP عمل می‌کند.
#include // Use UIPEthernet library for ENC28J60 // MAC address for your Ethernet module (check sticker or use a random one) byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED }; // IP address for the server (must be within your network’s range) IPAddress ip(192, 168, 1, 177); // Change to match your network // Initialize the Ethernet server on port 80 EthernetServer server(80); 
در بخش زیر (تابع راه‌اندازی):
•	Serial.begin(9600) برای چاپ لاگ‌های دیباگ روی پورت سریال راه‌اندازی می‌شود.
•	Ethernet.begin(mac, ip) استک شبکه را با MAC و IP مشخص راه‌اندازی می‌کند. (اگر بخواهید DHCP استفاده کنید باید فقط Ethernet.begin(mac) یا Ethernet.begin(mac) با گزینه DHCP فراخوانی کنید؛ در این کد IP به‌صورت استاتیک ست شده.)
•	server.begin() Listener سرور را فعال می‌کند تا ارتباط‌های ورودی را بپذیرد.
•	Ethernet.localIP() آدرسی را که ماژول در حال حاضر دارد چاپ می‌کند — مفید برای اطمینان از آدرس‌دهی صحیح.
void setup() { Serial.begin(9600); // Start Ethernet with MAC and IP Ethernet.begin(mac, ip); // Start the server server.begin(); Serial.print("Server IP: "); Serial.println(Ethernet.localIP()); } 
در کد زیر (تابع حلقه اصلی):
•	server.available() در صورت وجود یک کلاینت جدید یک شیء EthernetClient برمی‌گرداند.
•	متغیر currentLineIsBlank برای شناسایی پایان بخش‌هدرهای درخواست HTTP استفاده می‌شود: وقتی یک خط خالی (blank line) دریافت شود یعنی هدرها تمام شده‌اند و زمان ارسال پاسخ است.
•	حلقه while (client.connected()) تا زمان باز بودن اتصال ادامه می‌دهد. درون حلقه هر کاراکتر از درخواست توسط client.read() خوانده می‌شود.
•	منطق if (c == '\n' && currentLineIsBlank) زمانی اجرا می‌شود که دو newline متوالی (یا newline بعد از carriage-return) رخ داده باشد؛ در این وضعیت سرور هدر HTTP را ارسال می‌کند و سپس بدنه HTML را می‌فرستد.
•	پس از ارسال محتوا break می‌شود تا اتصال بسته شود؛ بعد از حلقه delay(1) کوتاه داده و client.stop() اتصال را خاتمه می‌دهد.
•	چاپ رشته‌های HTML با client.println() باعث ارسال خط به خط پاسخ HTTP می‌شود.
void loop() { // Listen for incoming clients EthernetClient client = server.available(); if (client) { Serial.println("New client connected"); // HTTP request ends with a blank line bool currentLineIsBlank = true; while (client.connected()) { if (client.available()) { char c = client.read(); // End of HTTP request if (c == '\n' && currentLineIsBlank) { // Send HTTP response header client.println("HTTP/1.1 200 OK"); client.println("Content-Type: text/html"); client.println("Connection: close"); client.println(); // Send HTML page client.println(""); client.println(""); client.println("Arduino Server"); client.println(""); client.println("

Hello World from Arduino!

"); client.println(""); client.println(""); break; } if (c == '\n') { currentLineIsBlank = true; } else if (c != '\r') { currentLineIsBlank = false; } } } // Give the client time to receive the data delay(1); client.stop(); Serial.println("Client disconnected"); } }
